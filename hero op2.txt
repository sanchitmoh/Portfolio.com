'use client'
import { motion } from 'framer-motion'
import { useState } from 'react'
import dynamic from 'next/dynamic'

const Scene3D = dynamic(() => import('./Scene3D'), { ssr: false })

      y: (e.clientY - rect.top - rect.height / 2) / 30
    })
  }

      id="home"

              opacity: [0, 0.6, 0],
              scale: [0, 1.5, 0]
            }}
            transition={{
              duration: Math.random() * 4 + 3,
              repeat: Infinity,
              delay: Math.random() * 3
            }}

          initial={{ opacity: 0, y: 50 }}

          style={{
            transform: `translate(${mousePosition.x}px, ${mousePosition.y}px)`

          {/* Professional name with 3D effect */}
          <motion.h1
            className="text-4xl xs:text-5xl sm:text-6xl md:text-7xl lg:text-8xl xl:text-9xl font-bold mb-4 sm:mb-6 lg:mb-8"
            initial={{ opacity: 0, scale: 0.8, rotateX: -15 }}
            animate={{ opacity: 1, scale: 1, rotateX: 0 }}
            transition={{ 
              duration: 1.8, 
              delay: 0.6, 
              type: "spring",
              stiffness: 80
            }}
          >
            <motion.span 
              className="text-off-white block"
              whileHover={{ 
                scale: 1.05,
                textShadow: "0 0 20px rgba(248, 250, 252, 0.3)"
              }}

              Sanchit

            animate={{ opacity: 1, y: 0, rotateY: 0 }}
            transition={{ duration: 1.2, delay: 1 }}
            className="mb-8 sm:mb-12 lg:mb-16 relative"
          >
            {/* Top underline */}
            <motion.div
              initial={{ width: 0 }}
              animate={{ width: "100%" }}

              className="h-0.5 bg-gradient-to-r from-transparent via-light-gray to-transparent mb-4 max-w-md mx-auto"
